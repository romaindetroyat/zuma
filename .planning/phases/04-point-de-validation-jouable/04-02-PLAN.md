```ts
export interface Ball { id: number; color: number; s: number; destroyingUntil?: number; power?: PowerKind; powerUntil?: number; comboDepth?: number }
export interface EngineConfig { tickHz; width; height; ballRadius; ballSpacing; colors; start; total; target; speed; feedSpeed; drainSpeed; projectileSpeed; accuracySpeedFactor; fireCooldownTicks; destroyTicks; attractSpeed; recoilSpeed; recoilTicks; slowFactor; slowTicks; backwardsSpeed; backwardsTicks; accuracyTicks; bombRadius; powerChance; powerGuardTicks; powerVisibleTicks; lives; levelBonus }
export const CANDIDATE: EngineConfig = { tickHz: 60, ballRadius: 16, ballSpacing: 32, projectileSpeed: 12, destroyTicks: 40, attractSpeed: 240, recoilSpeed: 90, recoilTicks: 8, ... };
const CONTACT_EPS = 0.5;
export class ZumaEngine {
  insert(index, ball: { id; color; power? }, _pos: Point, s: number, chain = this.chains[0]): Ball // splice + resolveForward(alive) — API de test, DOIT rester instantanée
  private shoot(x, y) // projectile x = frog.x + (dx/len) * ballRadius  ← devient muzzleOffset
  private componentIds(alive): number[] // coupure si alive[i].s - alive[i-1].s > ballSpacing + CONTACT_EPS
  private resolveForward(alive) // alive[i].s = max(alive[i].s, alive[i-1].s + spacing)
  private resolveBackward(alive, from): number
  private moveChain(chain) // before = componentIds; recul; poussoir; attraction (room = rear.s - (prevFront.s + spacing)); resolveForward; after = componentIds; onJunction(i) si fusion
  private onJunction(chain, alive, i) // emit merge; runAround; destroyRun(depth)
  private runAround(alive, index): Ball[] // touching(a,b) = b.s - a.s <= spacing + CONTACT_EPS
  private destroyRun(chain, alive, run, depth, shotId)
  private land(p: Projectile, hit: Ball, chain) // insert(...) ; emit insert ; gap bonus ; runAround → hits/streak/destroyRun/streak bonus ; else streak = 0
}
```
Ordre d'un `step()` : processCommands → tickPowers → spawnBalls → moveChain (par chaîne) → moveProjectiles (→ land) → processRemovals → maybeSpawnPower → checkEnd.
