// src/progress.ts — pure, DOM-free.
export interface RunState { stage: number; level: number; score: number; lives: number }   // stage / level 0-based, values at level start
export type WinRoute =
  | { kind: 'next-level'; run: RunState }                                                   // same stage, no screen
  | { kind: 'stage-done'; run: RunState; completed: number; temple: 1 | 2 | 3 | 4 | null }  // run = first level of the next stage
  | { kind: 'final'; temple: 5 };                                                           // last stage of the progression (index ≥ 12)
/** sizes = levels per stage (pack.data.stages[i].levels.length, today 5,5,5,6,6,6,7,7,7,7,7,7,1). score / lives = values at the end of the won level. */
export function afterWin(run: RunState, score: number, lives: number, sizes: readonly number[]): WinRoute;
/** Temple whose last stage is `completed`: 2 → 1, 5 → 2, 8 → 3, 11 → 4, last stage → 5, otherwise null (0x4317ec–0x43180e). */
export function templeAfter(completed: number, stageCount: number): 1 | 2 | 3 | 4 | 5 | null;
export const DOOR_COUNT = 12;
/** Door images ADVDOOR{temple}{letter}: stage 0..11 → temple ⌊stage / 3⌋ + 1, letter 'ABC'[stage % 3]. */
export function doorTemple(stage: number): 1 | 2 | 3 | 4;
export function doorLetter(stage: number): 'A' | 'B' | 'C';
/** One message key per door, push order 0x4422a0–0x442400 (A2, MEDIUM): index = min(stage, 11). */
export const DOOR_MESSAGES: readonly string[];  // DO_YOU_DARE_ENTER_THE_ANCIENT_RUINS … ENTER_THE_RUINS (12)
/** God dialogue per temple (A3, ASSUMED): god text keys shown in order, reply = [ribbit key, translation key]. */

export function insertScore(list: readonly HighScore[], entry: HighScore): { list: HighScore[]; rank: number | null };
/** Defensive read of the stored JSON (null / corrupt → defaultScores()); keeps ≤ 10 valid entries, names sanitized, finite non-negative integers. */
export function parseScores(raw: string | null): HighScore[];

export interface Profile { name: string; maxStage: number }       // maxStage: highest unlocked stage index (0..stageCount − 1)
export const PROFILE_KEY = 'zuma.profile.v1';
/** Trim, collapse inner spaces, drop control / non-printable characters, at most 16 characters; empty → null. */
export function sanitizeName(raw: unknown): string | null;
export function parseProfile(raw: string | null): Profile | null;
export const SAVE_KEY = 'zuma.save.v1';                            // RunState JSON
/** null when absent, corrupt, out of `sizes` bounds, lives < 1 or > 99, score negative or not finite. */
export function parseSave(raw: string | null, sizes: readonly number[]): RunState | null;
export interface StorageLike { getItem(key: string): string | null; setItem(key: string, value: string): void; removeItem(key: string): void }
/** try/catch around storage: missing storage or a throwing call → null / false, never an exception. */
export function readJson(storage: StorageLike | null, key: string): string | null;
export function writeJson(storage: StorageLike | null, key: string, value: unknown): boolean;   // value null → removeItem

export type AppScreen = 'welcome' | 'adventure' | 'level' | 'temple' | 'ending';
export interface AppState {
  screen: AppScreen;
  prompt: 'name' | 'continue' | null;          // modal dialogs over the welcome screen
  profile: Profile | null;
  scores: HighScore[];
  run: RunState | null;                        // current run (saved at every level start), null when none
  selected: number;                            // selected door / stage on the adventure screen
  opening: number | null;                      // door opening animation on arrival (stage index), else null

  | { kind: 'retry'; lives: number }                               // a life lost, lives remain: same level again
  | { kind: 'gameover'; score: number; date: string }              // engine reached gameover (submission, 0x4283c4)
  | { kind: 'leave' }                                              // gameover STATS closed, or « Retour au menu principal » confirmed
  | { kind: 'temple-done' } | { kind: 'ending-done' };
export type AppEffect =
  | { kind: 'start-level'; run: RunState }
  | { kind: 'save-run'; run: RunState | null }                     // null = clear the save
  | { kind: 'save-profile'; profile: Profile }
  | { kind: 'save-scores'; scores: HighScore[] }
  | { kind: 'sound'; key: 'templecomplete' | 'button' };
/** Boot: review run (from ?level=) → screen 'level' + start-level effect; else 'welcome', prompt 'name' when no profile. */
export function initialApp(input: { profile: Profile | null; save: RunState | null; scores: HighScore[]; sizes: readonly number[]; startLives: number; review: RunState | null }): { state: AppState; effects: AppEffect[] };
/** Pure transition; unknown or out-of-place events return the same state and no effect. */
export function reduceApp(state: AppState, event: AppEvent): { state: AppState; effects: AppEffect[] };
